function validaDiaMes(v, ano){

  //Remove tudo o que nÃ£o Ã© dÃ­gito
  v=v.replace(/\D/g,"")

  // Valida a quantidade de caracteres
  if(v.length != 4 && v.length != 8){
    return "Data InvÃ¡lida";
  }

  // Valida o Mes
  var mes = v.substring(2,4);
  if(mes > 12) {

    return "MÃªs InvÃ¡lido";
  }

  qtdDia = 31;

  switch(mes) {
    case '02':

      qtdDia = 28;
      if (ano % 400 == 0) {

        qtdDia = 29;
      } else {

        if(ano%4 == 0 && ano%100!=0) {

          qtdDia = 29;
        }
      }
    break;
    case '04':

      qtdDia = 30;
    break;
    case '06':

      qtdDia = 30;
    break;
    case '09':

      qtdDia = 30;
    break;
    case '11':

      qtdDia = 30;
    break;
    default:
      qtdDia = 31;
  }

  // Valida o dia
  if(v.substring(0,2) > qtdDia) {

    return "Dia InvÃ¡lido, o mÃªs selecionado possui no mÃ¡ximo " + qtdDia + " dias";
  }

  return true;

}

function ComparaData(data1, data2){

  var oResult = new Object();
    oResult.inicio = "";
    oResult.fim = "";

  var sInicio  = $('#'+data1).val();
  var sFim     = $('#'+data2).val();

  data1 = sInicio.replace(/\D/g,"")
  data2 = sFim.replace(/\D/g,"")

  var mes1 = data1.substring(2,4);
  var mes2 = data2.substring(2,4);

  var ano1 = data1.substring(4,8);
  var ano2 = data2.substring(4,8);

  bInicio = validaDiaMes(sInicio, ano1);
  bFim    = validaDiaMes(sFim, ano2);

  // Compara se o ano1 Ã© menor(anterior) ou igual ao ano2
  // Compara se o mes1 Ã© menor (anterior) ao mes2
  // se for ja retorna true, pois foi validada a comparacao
  // caso contrario efetua comparacao de mes igual e depois o dia
  // caso mes1 seja maior que mes2, e retornado false indicando a data inferior

  if(bInicio != true){

    alert("Data inicial: " + bInicio);
    return false;
  }

  if(bFim != true){

    alert("Data final: " + bFim);
    return false;
  }

  if(ano1 < ano2){ //ano inicial inferior

    oResult.inicio  = sInicio;
    oResult.fim     = sFim;
  } else {

    if(ano1 == ano2){ //mesmo ano

      if(mes1 < mes2){ //mes inferior

        oResult.inicio  = sInicio;
        oResult.fim     = sFim;
      } else {

        if(mes1 == mes2){ //mesmo mes

          var dia1 = data1.substring(0,2);
          var dia2 = data2.substring(0,2);

          if(dia1 <= dia2){ //dia tem que ser menor ou igual

            oResult.inicio  = sInicio;
            oResult.fim     = sFim;
          } else {

            alert('Dia final inferior ao mes inicial para o ano e mes selecionado');
            return false;
          }
        } else {

          alert('Mes final inferior ao mes inicial para o ano selecionado');
          return false;
        }
      }
    } else {

      alert('Ano Final inferior ao ano Inicial')
      return false;
    }
  }
  // return false;
  return oResult;
}
